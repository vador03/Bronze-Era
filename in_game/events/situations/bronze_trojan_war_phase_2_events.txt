namespace = start_trojan_war_phase_2

start_trojan_war_phase_2.1 = {
    type = country_event
    title = trojan_war_phase_2_start.t
    desc = trojan_war_phase_2_start.d
    is_triggered_only = yes
    hidden = no

immediate = {
    declare_war_with_cb = {
        target = c:WILUS
        type = casus_belli:trojan_war_casus_belli
        set_variable = {
            name = is_trojan_war
            value = yes
        }
    }
    c:WILUS = { save_scope_as = target_country }
    c:0001G = { save_scope_as = target }
    random_current_war = {
        limit = {
            any_defender = { this = c:WILUS }
        }
        save_scope_as = trojan_war 
    }
    c:AHHI = { trigger_event_non_silently = { id = join_trojan_War.1 days = 10 } }
    c:LYCIA = { trigger_event_non_silently = { id = join_trojan_War.1 days = 10 } }
    c:SEHA = { trigger_event_non_silently = { id = join_trojan_War.1 days = 10 } }
    c:LUKKA = { trigger_event_non_silently = { id = join_trojan_War.1 days = 10 } }
    c:MASA = { trigger_event_non_silently = { id = join_trojan_War.1 days = 10 } }
} 
option = {
    name = "trojan_war_phase_2_start.a"
} 
}

namespace = trojan_war_end

trojan_war_end.1 = {
    type = country_event
    title = trojan_war_end_01.t
    desc = trojan_war_end_01.d
    is_triggered_only = yes
    hidden = no
# For Greece victory
    immediate = {
        root = { 
                add_prestige = 10
            OR = {
                if = { limit = { global_var:anatolia_raid_success_count = 1 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_one  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count = 2 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_two  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count = 3 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_three  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count = 4 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_four  
                    days = -1
                    }
                }
                if = { limit = { global_var:anatolia_raid_success_count >= 5 } 
                    add_country_modifier = { 
                    mode = add
                    modifier = trojan_war_end_greece_victory_count_five  
                    days = -1
                    }
                }
            }
        }
    }
    option = {
        name = "trojan_war_end.a"
    }
}

namespace = bronze_unserved_religions

# Two contextual stories per religion; one country-wide cooldown keeps them rare.

bronze_unserved_religions.1 = {
	type = country_event
	title = bronze_unserved_religions.1.title
	desc = bronze_unserved_religions.1.desc
	outcome = neutral
	trigger = {
		is_real_country = yes
		religion = religion:bilozerka_rites
		NOT = { has_variable = be_religion_story_cooldown }
		current_date <= 410.1.1
		any_owned_location = { province = { province_food_percentage < 0.25 } }
	}
	immediate = { set_variable = { name = be_religion_story_cooldown years = 6 } }
	option = {
		name = bronze_unserved_religions.1.a
		trigger = { gold >= 15 }
		ai_chance = { factor = 55 }
		add_gold = -15
		add_stability = 2
		add_country_modifier = { modifier = be_rite_pasture years = 3 }
	}
	option = {
		name = bronze_unserved_religions.1.b
		ai_chance = { factor = 45 }
		add_prestige = 1
		add_country_modifier = { modifier = be_rite_shelter years = 2 }
	}
}

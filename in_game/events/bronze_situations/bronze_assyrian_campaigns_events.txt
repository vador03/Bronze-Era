	title = bronze_assyrian_campaigns.1.title

	outcome = neutral
	major = yes
	orphan = yes
	trigger = { tag = ASYRI }
	option = {
		name = bronze_assyrian_campaigns.1.a

			limit = { location:damascus = { has_owner = yes NOT = { owner = root } } }
			add_casus_belli = { type = casus_belli:cb_conquer_province province = location:damascus.province target = location:damascus.owner years = 30 }
		}
		if = {
			limit = { location:gaza = { has_owner = yes NOT = { owner = root } } }
			add_casus_belli = { type = casus_belli:cb_conquer_province province = location:gaza.province target = location:gaza.owner years = 30 }
		}
		add_gold = 75
		add_prestige = prestige_weak_bonus
	}
}

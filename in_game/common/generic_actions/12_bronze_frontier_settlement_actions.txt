# The first picker can be supplied by a clicked vacant location in the GUI.
# Otherwise the standard generic-action map picker selects one location.
bronze_begin_frontier_settlement = {
	type = owncountry
	icon = migrate_pop_based_country
	show_message = no
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	price = { value = 25 }

	potential = {
		scope:actor = { bronze_frontier_settlement_unlocked = yes }
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = bronze_frontier_settlement_target }
			is_during_bankruptcy = no
			gold >= 25
		}
	}

	select_trigger = {
		looking_for_a = location
		source = actor
		source_flags = vacant_adjacent_locations
		target_flag = target_location
		name = bronze_frontier_choose_target
		none_available_msg_key = bronze_frontier_no_target
		column = { data = name }
		column = { data = population }
		visible = {
			bronze_frontier_settlement_target_allowed = { country = scope:actor }
			NOT = { has_variable = bronze_frontier_settlement_country }
		}
	}

	select_trigger = {
		looking_for_a = location
		source = actor
		target_flag = source_location
		name = bronze_frontier_choose_source
		none_available_msg_key = bronze_frontier_no_source
		column = { data = name }
		column = { data = population }
		visible = {
			bronze_frontier_settlement_source_allowed = {
				country = scope:actor
				target = scope:target_location
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target_location
				exists = scope:source_location
				scope:target_location = {
					bronze_frontier_settlement_target_allowed = { country = scope:actor }
					NOT = { has_variable = bronze_frontier_settlement_country }
				}
				scope:source_location = {
					bronze_frontier_settlement_source_allowed = {
						country = scope:actor
						target = scope:target_location
					}
				}
			}
			scope:actor = {
				set_variable = { name = bronze_frontier_settlement_target value = scope:target_location }
				set_variable = { name = bronze_frontier_settlement_source value = scope:source_location }
			}
			scope:target_location = {
				set_variable = { name = bronze_frontier_settlement_country value = scope:actor }
				set_variable = { name = bronze_frontier_settlement_progress value = 0 }
				set_variable = { name = bronze_frontier_settlement_months_left value = 40 }
			}
		}
	}

	ai_will_do = { add = 10 }
}

bronze_cancel_frontier_settlement = {
	type = owncountry
	icon = migrate_pop_based_country
	show_message = no
	ai_tick = never
	automation_tick = never

	potential = {
		scope:actor = { has_variable = bronze_frontier_settlement_target }
	}
	allow = { scope:actor = { has_variable = bronze_frontier_settlement_target } }
	effect = { scope:actor = { bronze_frontier_settlement_clear = yes } }
}

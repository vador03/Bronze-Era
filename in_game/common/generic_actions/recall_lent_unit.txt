recall_lent_unit = {
	type = owncountry
	

	potential = {
		scope:actor = {
			any_rented_out_mercenary = { is_lent_unit = yes }
		}
	}

	allow = {
	}

	ai_tick = monthly
	ai_tick_frequency = 6

	# Pick which lent unit to recall.
	# The default mercenary enumeration only surfaces units a country has put up for hire
	# (_MercenariesForHire); lent units are never added there, so the candidate list must be
	# built explicitly from the units this country has lent out.
	select_trigger = {
		looking_for_a = mercenary
		target_flag = lent_merc_target
		interaction_source_list = {
			scope:actor = {
				every_rented_out_mercenary = {
					limit = {
						is_lent_unit = yes
						# Skip embarked units — RecallLentUnit() destroys and recreates the
						# underlying unit, which would tear the embark link and drown loaded armies.
						NOT = { is_carrying_or_loaded_unit = yes }
					}
					add_to_list = source
				}
			}
		}
		cache_interaction_source_list = yes
		name = "recall_lent_unit_choose_unit"
		none_available_msg_key = "recall_lent_unit_no_units"
		column = { data = name }
		visible = {
			is_lent_unit = yes
			NOT = { is_carrying_or_loaded_unit = yes }
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:lent_merc_target
			}
			# Give opinion hit to the borrower if a shared war is still ongoing
			scope:lent_merc_target = {
				customer = {
					save_temporary_scope_as = borrower
				}
			}
			if = {
				limit = {
					exists = scope:borrower
					scope:actor = {
						any_current_war = {
							is_in_war = scope:borrower
						}
					}
				}
				scope:borrower = {
					add_opinion = {
						target = scope:actor
						modifier = opinion_recalled_lent_unit
					}
				}
			}
			scope:lent_merc_target = {
				recall_lent_unit = {}
			}
		}
	}

	ai_will_do = {
		value = 0

		# Recall when the lender is under serious threat
		if = {
			limit = {
				scope:actor = {
					at_war = yes
					lowest_war_score < -15
				}
			}
			add = 100
		}
	}
}

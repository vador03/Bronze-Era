# The most recent eligible king is installed once. This also lets a Rome
# founded later than 753 BC enter the appropriate part of the sequence.
bronze_rome_install_king_for_date = {
	if = {
		limit = { current_date >= 675.1.1 current_date < 701.1.1 var:bronze_rome_king_stage < 7 }
		bronze_rome_install_superbus = yes
	}
	else_if = {
		limit = { current_date >= 635.1.1 current_date < 675.1.1 var:bronze_rome_king_stage < 6 }
		bronze_rome_install_servius = yes
	}
	else_if = {
		limit = { current_date >= 594.1.1 current_date < 635.1.1 var:bronze_rome_king_stage < 5 }
		bronze_rome_install_priscus = yes
	}
	else_if = {
		limit = { current_date >= 569.1.1 current_date < 594.1.1 var:bronze_rome_king_stage < 4 }
		bronze_rome_install_ancus = yes
	}
	else_if = {
		limit = { current_date >= 537.1.1 current_date < 569.1.1 var:bronze_rome_king_stage < 3 }
		bronze_rome_install_tullus = yes
	}
	else_if = {
		limit = { current_date >= 495.1.1 current_date < 537.1.1 var:bronze_rome_king_stage < 2 }
		bronze_rome_install_numa = yes
	}
}

bronze_rome_adopt_republic = {
	remove_variable = bronze_rome_republic_pending
	set_variable = { name = bronze_rome_republic_resolved value = 1 }
	set_variable = { name = bronze_rome_king_stage value = 8 }
	change_government_type = government_type:republic
	kill_character_silently = ruler
}

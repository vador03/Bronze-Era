# Assyrian kings from 1202 to 912 BC. Engine year = 1210 - displayed BC year.
# Ages and abilities are gameplay estimates; accession dates follow the supplied list.
bronze_assyria_install_adaside_king = {
	create_character = {
		first_name = $name$
		dynasty = adaside_dynasty
		culture = culture:assyrian
		religion = religion:assyrian
		birth_location = location:mosul
		estate = estate_type:nobles_estate
		female = no
		age = $age$
		adm = { 40 70 }
		dip = { 40 70 }
		mil = { 40 70 }
		save_scope_as = bronze_assyria_new_king
	}
	set_new_ruler = scope:bronze_assyria_new_king
	set_variable = { name = bronze_assyria_king_stage value = $stage$ }
	set_variable = { name = $id$ value = scope:bronze_assyria_new_king }
}

# Country scope. Called only after the decision has selected a valid escort.
be_migration_colonize_effect = {
	if = {
		limit = {
			has_tribal_government = yes
			country_type = army
			has_variable = be_migration_active
			has_variable = be_migration_camp
			NOT = { has_variable = be_migration_colony_cooldown }
			at_war = no
			gold >= 25
			num_locations < 5
			exists = scope:be_colony_unit
			scope:be_colony_unit = { be_migration_colony_unit_allowed = { country = root } }
		}
		scope:be_colony_unit.unit_location = {
			save_scope_as = be_colony_location
			change_location_owner = root
			change_integration_level = colonized
			change_location_controller = root
		}
		if = {
			limit = { var:be_migration_camp != scope:be_colony_location }
			var:be_migration_camp = {
				every_pop = {
					limit = { owner = root pop_size > 0.01 }
					transfer_pop = {
						target = scope:be_colony_location
						operator = multiply
						value = 0.20
					}
				}
			}
		}
		add_gold = -25
		set_variable = { name = be_migration_colony_cooldown value = 1 years = 5 }
	}
}

# Country scope. Turns the army-country into a territorial country at its escort.
be_migration_settle_effect = {
	if = {
		limit = {
			has_tribal_government = yes
			country_type = army
			has_variable = be_migration_active
			at_war = no
			exists = scope:be_colony_unit
			scope:be_colony_unit = { be_migration_settle_unit_allowed = { country = root } }
		}
		scope:be_colony_unit.unit_location = {
			if = {
				limit = { has_owner = no }
				change_location_owner = root
				change_integration_level = colonized
				change_location_controller = root
			}
		}
		if = {
			limit = {
				has_variable = be_migration_camp
				var:be_migration_camp != scope:be_colony_unit.unit_location
			}
			var:be_migration_camp = {
				every_pop = {
					limit = { owner = root }
					transfer_pop = {
						target = scope:be_colony_unit.unit_location
						operator = multiply
						value = 1.00
					}
				}
			}
		}
		set_capital = scope:be_colony_unit.unit_location
		change_country_type = location
		remove_variable = be_migration_active
		remove_variable = be_migration_camp
		remove_variable = be_migration_passage_handled
	}
}

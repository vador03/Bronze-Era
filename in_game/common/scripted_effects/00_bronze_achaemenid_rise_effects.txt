# Situation scope. Rebuild the small pool yearly so a defeated Median realm can
# give way to a stronger Iranian successor without leaving a stale claimant.
bronze_achaemenid_select_claimant = {
	clear_variable_list = bronze_achaemenid_candidate_pool
	every_country = {
		limit = { bronze_achaemenid_rise_candidate = yes }
		remove_variable = bronze_achaemenid_claimant
		save_scope_as = bronze_achaemenid_candidate_scope
		root = {
			add_to_variable_list = {
				name = bronze_achaemenid_candidate_pool
				target = scope:bronze_achaemenid_candidate_scope
			}
		}
	}
	ordered_in_list = {
		variable = bronze_achaemenid_candidate_pool
		order_by = {
			value = country_total_army_levy_size
			add = { value = army_size multiply = 10 }
			add = { value = monthly_income_trade_and_tax multiply = 10 }
			add = { value = country_rank_level multiply = 20 }
			# The historical Median head start ends around 700 BC.
			if = {
				limit = { tag = MEDIA current_date < 510.1.1 }
				add = 100
			}
		}
		max = 1
		check_range_bounds = no
		set_variable = { name = bronze_achaemenid_claimant value = 1 }
		if = {
			limit = { is_ai = yes }
			set_personality = ai_personality:ai_expansionist
		}
		if = {
			limit = { NOT = { has_variable = bronze_achaemenid_claimant_announced } }
			set_variable = { name = bronze_achaemenid_claimant_announced value = 1 }
			trigger_event_non_silently = bronze_achaemenid_rise.1
		}
	}
}

# All campaign counters live on the situation. Persian lands and satrapies both
# count, so delegating a conquest cannot erase progress toward an objective.
bronze_achaemenid_refresh_campaign_progress = {
	set_variable = { name = bronze_achaemenid_campaign_current value = 0 }
	if = {
		limit = { var:bronze_achaemenid_campaign_stage = 1 }
		c:PERSIA ?= {
			every_owned_location = {
				limit = { area = area:iraq_arabi_area }
				situation:bronze_achaemenid_rise = {
					change_variable = { name = bronze_achaemenid_campaign_current add = 1 }
				}
			}
			every_subject = {
				limit = { is_subject_type = bronze_satrapy }
				every_owned_location = {
					limit = { area = area:iraq_arabi_area }
					situation:bronze_achaemenid_rise = {
						change_variable = { name = bronze_achaemenid_campaign_current add = 1 }
					}
				}
			}
		}
	}
	else_if = {
		limit = { var:bronze_achaemenid_campaign_stage = 2 }
		c:PERSIA ?= {
			every_owned_location = {
				limit = {
					OR = {
						area = area:eastern_khorasan_area
						area = area:sistan_area
					}
				}
				situation:bronze_achaemenid_rise = {
					change_variable = { name = bronze_achaemenid_campaign_current add = 1 }
				}
			}
			every_subject = {
				limit = { is_subject_type = bronze_satrapy }
				every_owned_location = {
					limit = {
						OR = {
							area = area:eastern_khorasan_area
							area = area:sistan_area
						}
					}
					situation:bronze_achaemenid_rise = {
						change_variable = { name = bronze_achaemenid_campaign_current add = 1 }
					}
				}
			}
		}
	}
	set_variable = { name = bronze_achaemenid_campaign_gains value = var:bronze_achaemenid_campaign_current }
	change_variable = { name = bronze_achaemenid_campaign_gains subtract = var:bronze_achaemenid_campaign_baseline }
	if = {
		limit = { var:bronze_achaemenid_campaign_gains < 0 }
		set_variable = { name = bronze_achaemenid_campaign_gains value = 0 }
	}
}

bronze_achaemenid_start_western_campaign = {
	set_variable = { name = bronze_achaemenid_empire_formed value = 1 }
	set_variable = { name = bronze_achaemenid_campaign_stage value = 1 }
	set_variable = { name = bronze_achaemenid_campaign_months_left value = 240 }
	set_variable = { name = bronze_achaemenid_campaign_baseline value = 0 }
	set_variable = { name = bronze_achaemenid_campaign_gains value = 0 }
	remove_variable = bronze_achaemenid_campaign_resolution_pending
	c:PERSIA ?= {
		add_area_preference = bronze_achaemenid_conquer_babylonia
		if = { limit = { is_ai = yes } set_personality = ai_personality:ai_expansionist }
		every_owned_location = {
			limit = { area = area:iraq_arabi_area }
			situation:bronze_achaemenid_rise = {
				change_variable = { name = bronze_achaemenid_campaign_baseline add = 1 }
			}
		}
		every_subject = {
			limit = { is_subject_type = bronze_satrapy }
			every_owned_location = {
				limit = { area = area:iraq_arabi_area }
				situation:bronze_achaemenid_rise = {
					change_variable = { name = bronze_achaemenid_campaign_baseline add = 1 }
				}
			}
		}
		if = {
			limit = {
				location:hillah = {
					has_owner = yes
					NOT = { owner = c:PERSIA }
					owner ?= { NOT = { is_subject_of = c:PERSIA } }
				}
			}
			add_casus_belli = {
				type = casus_belli:cb_conquer_province
				province = location:hillah.province
				target = location:hillah.owner
				years = 20
			}
		}
	}
	set_variable = { name = bronze_achaemenid_campaign_current value = var:bronze_achaemenid_campaign_baseline }
}

bronze_achaemenid_start_eastern_campaign = {
	set_variable = { name = bronze_achaemenid_campaign_stage value = 2 }
	set_variable = { name = bronze_achaemenid_campaign_months_left value = 240 }
	set_variable = { name = bronze_achaemenid_campaign_baseline value = 0 }
	set_variable = { name = bronze_achaemenid_campaign_gains value = 0 }

		add_area_preference = bronze_achaemenid_conquer_eastern_marches
		every_owned_location = {
			limit = {
				OR = {
					area = area:eastern_khorasan_area
					area = area:sistan_area

			}
			situation:bronze_achaemenid_rise = {
				change_variable = { name = bronze_achaemenid_campaign_baseline add = 1 }
			}
		}
		every_subject = {
			limit = { is_subject_type = bronze_satrapy }
			every_owned_location = {
				limit = {
					OR = {
						area = area:eastern_khorasan_area
						area = area:sistan_area
					}
				}
				situation:bronze_achaemenid_rise = {
					change_variable = { name = bronze_achaemenid_campaign_baseline add = 1 }
				}
			}
		}
		if = {
			limit = {
				location:balkh = {

bronze_achaemenid_finish_campaigns = {
	set_variable = { name = bronze_achaemenid_campaign_stage value = 3 }
	remove_variable = bronze_achaemenid_campaign_resolution_pending
	c:PERSIA ?= { remove_area_preference = bronze_achaemenid_conquer_eastern_marches }
}

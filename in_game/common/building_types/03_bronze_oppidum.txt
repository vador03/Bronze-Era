# Iron Age hilltop settlement defense. Culture access is repeated here so
# inherited technologies cannot open oppida to unrelated countries.
bronze_age_oppidum = {
	is_foreign = no
	max_levels = 1
	pop_type = soldiers
	forbidden_for_estates = yes
	category = defense_category
	employment_size = stockade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	country_potential = {
		OR = {
			culture = { has_culture_group = culture_group:celtic }
			culture = { has_culture_group = culture_group:gaulish_bronze_group }
			culture = { has_culture_group = culture_group:germanic_nordic_group }
			culture = { has_culture_group = culture_group:britain_bronze_group }
		}
	}
	location_potential = { topography = hills }

	build_time = medium_fort_building
	expensive = yes
	possible_production_methods = { town_walls_maintenance }
	construction_demand = town_walls_construction

	modifier = {
		local_garrison_size = 0.35
		local_defensive = 0.15
		movement_cost = 0.1
	}

	# City walls already establish a fort level; this does not add another.
	raw_modifier = {
		minimum_fort_level = 1
		propagating_zone_of_control = yes
	}

	graphical_tags = { small_city_walls city_walls }
}

on_game_start = {
	on_actions = {
		bronze_chronology_game_start
	}
}

monthly_country_pulse = {
	on_actions = {
		bronze_chronology_monthly_pulse
		bronze_city_prestige_runtime_bootstrap
		bronze_city_prestige_player_rank_bootstrap
		bronze_city_prestige_player_rank_repair
		bronze_rome_foundation_monthly_pulse
		bronze_etruscan_cities_monthly_pulse
		bronze_carthage_foundation_monthly_pulse
		bronze_massalia_foundation_monthly_pulse
		bronze_yerevan_foundation_monthly_pulse
	}
}

bronze_chronology_game_start = {
	effect = {
		# Proper chronology shim:
		# The engine stays on stable positive years. These country variables drive
		# the topbar's visual Bronze Age date: internal year 1 = 1209 BC.
		every_country = {
			limit = {
				is_real_country = yes
			}
			set_variable = {
				name = bronze_display_year
				value = {
					value = 1210
					subtract = current_year
				}
			}
			set_variable = {
				name = bronze_display_ad_year
				value = {
					value = current_year
					subtract = 1209
					min = 0
				}
			}
			if = {
				limit = {
					var:bronze_display_year < 1
				}
				set_variable = { name = bronze_display_year value = 0 }
			}
		}

		if = {
			limit = {
				NOT = { has_global_variable = bronze_city_prestige_runtime_v9_initialized }
			}
			bronze_city_prestige_initialize_world = yes
		}
	}
}

bronze_chronology_monthly_pulse = {
	trigger = {
		is_real_country = yes
	}
	effect = {
		# Recalculate from current_year instead of decrementing from a pulse.
		# This prevents the visual year from changing early when yearly_country_pulse
		# happens on a country-specific date rather than on January 1.
		set_variable = {
			name = bronze_display_year
			value = {
				value = 1210
				subtract = current_year
			}
		}
		set_variable = {
			name = bronze_display_ad_year
			value = {
				value = current_year
				subtract = 1209
				min = 0
			}
		}
		if = {
			limit = {
				var:bronze_display_year < 1
			}
			set_variable = { name = bronze_display_year value = 0 }
		}
	}
}

bronze_rome_foundation_monthly_pulse = {
	trigger = {
		bronze_rome_foundation_candidate = yes
	}

	effect = {
		if = {
			limit = { is_human = yes }
			set_global_variable = { name = bronze_rome_foundation_event_pending value = 1 }
			if = {
				limit = { bronze_rome_foundation_latin_owner = yes }
				trigger_event_non_silently = bronze_rome_foundation.1
			}
			else = {
				trigger_event_non_silently = bronze_rome_foundation.2
			}
		}
		else = {
			bronze_rome_foundation_create_kingdom = yes
		}
	}
}

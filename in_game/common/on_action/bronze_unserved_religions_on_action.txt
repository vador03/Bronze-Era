be_rite_pulse_bilozerka_rites = {
	trigger = {
		is_real_country = yes
		religion = religion:bilozerka_rites
		NOT = { has_variable = be_religion_story_cooldown }
		current_date <= 410.1.1
	}
	effect = {
		if = {
			limit = { any_owned_location = { province = { province_food_percentage < 0.25 } } }
			random = { chance = 8 trigger_event_non_silently = { id = bronze_unserved_religions.1 } }
		}
		if = {
			limit = { NOT = { has_variable = be_religion_story_cooldown } at_war = yes }
			random = { chance = 8 trigger_event_non_silently = { id = bronze_unserved_religions.2 } }
		}
	}
}

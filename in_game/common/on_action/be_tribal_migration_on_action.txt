monthly_country_pulse = {
	on_actions = { be_migration_quarterly_pulse }
}

# Only active migrant countries pay for a quarterly food check.
be_migration_quarterly_pulse = {
	trigger = {
		has_variable = be_migration_active
		OR = {
			current_month = 1
			current_month = 4
			current_month = 7
			current_month = 10
		}
	}
	events = { be_migration.1 be_migration.2 }
}
